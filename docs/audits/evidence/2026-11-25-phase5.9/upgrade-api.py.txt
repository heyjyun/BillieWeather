import json,sys,urllib.request,urllib.error,hashlib
B="http://127.0.0.1:"+sys.argv[1]
def call(m,p,tok=None,body=None):
    r=urllib.request.Request(B+p,method=m,data=json.dumps(body).encode() if body is not None else None,headers={"Accept":"application/json",**({"Content-Type":"application/json"} if body is not None else {}),**({"Authorization":"Bearer "+tok} if tok else {})})
    try:
        with urllib.request.urlopen(r,timeout=60) as x: d=x.read(); return x.status,(json.loads(d) if d else None)
    except urllib.error.HTTPError as e: return e.code,None
