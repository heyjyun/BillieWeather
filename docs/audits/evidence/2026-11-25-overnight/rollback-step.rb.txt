require "json"
state_file = ENV.fetch("STATE"); state = File.exist?(state_file) ? JSON.parse(File.read(state_file)) : {}
email = "rollback-fence@example.com"
user = User.find_by(email: email)
unless user
  user = User.new(email: email, password: "Synthetic-9k2-Violet-Harbor", first_name: "Q", last_name: "A", time_zone: "UTC", usage_region: "US", nickname: "QA")
  user.save!(validate: false)
end
def status(token)
  s = ActionDispatch::Integration::Session.new(Rails.application); s.https!; s.host! "localhost"
  s.get "/api/client/v1/account", headers: {"Authorization" => "Bearer #{token}"}; s.response.status
end
out = {}
(state["tokens"] || {}).each { |name, t| out[name] = status(t) }
if ENV["ISSUE"]
  (state["tokens"] ||= {})[ENV["ISSUE"]] = ClientApi::SessionToken.issue(user).first
  out["issued #{ENV["ISSUE"]}"] = status(state["tokens"][ENV["ISSUE"]])
end
if ENV["SIGNOUT"]
  s = ActionDispatch::Integration::Session.new(Rails.application); s.https!; s.host! "localhost"
  s.delete "/api/client/v1/session", headers: {"Authorization" => "Bearer #{state["tokens"][ENV["SIGNOUT"]]}"}
  out["signed out #{ENV["SIGNOUT"]}"] = [s.response.status, status(state["tokens"][ENV["SIGNOUT"]])]
end
File.write(state_file, JSON.generate(state)); puts "RESULT #{out.to_json}"
