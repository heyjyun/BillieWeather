accept.NEW = NEW

def front_guard():
    f = subprocess.run([S + "/idle"], capture_output=True, text=True).stdout
    if "front BillieWeather QA p63" not in f: raise Stop("PAUSED: " + f.strip())
def signin_or_home(t=20):
    end = time.time() + t
    while time.time() < end:
        x = dump()
        if "AXButton 'All Transactions'" in x: return "still signed in"
        if "Sign in to BillieWeather" in x: login(); return "signed in again"
        time.sleep(.5)
    front_guard(); raise RuntimeError("neither home nor sign-in shown")

def open_new():
    quit() if running() else None
    expect(q("select max(version) from _sqlx_migrations") == [(10,)])
    launch2(NEW); return signin_or_home()

def cats():
    rules = q("select count(*) from merchant_rules")[0][0]; own = q("select count(*) from custom_categories")[0][0]
    press("Settings"); press("Open Feature Library"); wait("Coming soon")
    x = d("dump", "full"); hits = [l.strip() for l in x.splitlines() if "ategor" in l]
    for cand in ("~Show Categories & Tags", "~Categories & Tags"):
        try: press(cand); break
        except RuntimeError: pass
    else: raise RuntimeError("switch not found: " + " | ".join(hits)[:250])
    time.sleep(1.5); press("~Categories", 0); x = wait("ZXQ Hobby")
    d("shot", S + "/shots/e2-categories.png")
    expect(q("select count(*) from merchant_rules")[0][0] == rules); expect(q("select count(*) from custom_categories")[0][0] == own)
    return f"own category shown; {rules} rule(s), {own} own categor(ies) unchanged"

def undo_smoke():
    manual_home(); press("~Transactions Search, filter"); wait("New entry")
    before = q("select id,note from transactions where note is not null or 1 order by id limit 1")
    press("Edit"); wait("Note ( optional )"); fill("Note ( optional )", "rc2 undo check"); press("Save"); time.sleep(1.5)

    lab = undo_label(); expect(lab and lab.startswith("Undo:"), str(lab)); press("~Undo:"); time.sleep(1.5)
    expect(q("select count(*) from transactions where note='rc2 undo check'") == [(0,)], "undo did not restore")
    d("shot", S + "/shots/e2-undo-bar.png"); return lab

def soft_quit():
    if not running(): return
    subprocess.run(["osascript", "-e", 'tell application id "com.billieweather.client.qa.p63" to quit'], capture_output=True)
    for _ in range(20):
        if not running(): return
        time.sleep(.5)
    for pid in running(): os.kill(pid, 15)   # QA bundles from this scratchpad only (pgrep "QA p63.app")
    time.sleep(3)
    if running(): raise RuntimeError("QA app did not quit")
def launch2(app):
    soft_quit(); launch(app)
def manual_home():
    if "AXButton 'Add & Sync'" not in dump(): press("~Manual"); time.sleep(1)
    press("Add & Sync"); press("Manual bookkeeping"); time.sleep(.8)

    shutil.copy(S + "/v9-book.db", DB()); c = sqlite3.connect(DB()); c.execute("CREATE TRIGGER history_budgets_delete BEFORE DELETE ON budgets BEGIN SELECT 1; END"); c.commit(); c.close(); b = book()
    launch2(NEW)
    end = time.time() + 25; x = ""

        x = dump()

        time.sleep(1)
    if "Sign in to BillieWeather" in x and "could not" not in x:
        login() if False else None

    expect(q("select count(*) from sqlite_master where name like 'history%' and name<>'history_budgets_delete'") == [(0,)])

CK = S + "/ckpt-e2-after-edit"   # QA data folder as it stood right after step 5 passed

    launch2(NEW); r = signin_or_home(); expect(q("select max(version) from _sqlx_migrations") == [(10,)]); sound(); return same(b, book()) + "; " + r

def oldrefuses():
    rows = book(); launch2(OLD); end = time.time() + 25; x = ""
    while time.time() < end:   # 0.1.0 may refuse before or after the sign-in screen
        x = dump()

    for s in ("", "-wal", "-shm"):
        if os.path.exists(DB() + s): shutil.move(DB() + s, aside + "/billieweather.db" + s)
    shutil.copy(bak, DB()); os.path.exists(bak + "-wal") and shutil.copy(bak + "-wal", DB() + "-wal"); os.path.exists(DB() + ".schema") and os.remove(DB() + ".schema")
    launch2(OLD); time.sleep(4); x = dump(); login() if "Sign in to BillieWeather" in x else wait("AXButton 'All Transactions'", 20)
    expect(q("select max(version) from _sqlx_migrations") == [(9,)]); expect(q("select count(*) from transactions where note='made in 0.9'") == [(0,)])
    return same(book(S + "/v9-book.db"), book()) + f" (from {os.path.basename(bak)})"
step("rollback as documented: the automatic copy opens in 0.1.0 with every row", back)

def forward():
    press("Settings"); press("Sign out"); wait("Sign in to BillieWeather"); time.sleep(1); soft_quit()
    # The only pre-v10 copy was taken at the first (deliberately failing) upgrade, so it still carries
    # the injected trigger. Remove the cause again exactly as in step 2, then upgrade.
    c = sqlite3.connect(DB()); injected = c.execute("select count(*) from sqlite_master where name='history_budgets_delete'").fetchone()[0]
    c.execute("DROP TRIGGER IF EXISTS history_budgets_delete"); c.commit(); c.close()
    launch2(NEW); wait("Sign in to BillieWeather", 20); login()
    expect(q("select max(version) from _sqlx_migrations") == [(10,)]); sound(); r = same(book(S + "/v9-book.db"), book()); soft_quit()
    return r + f"; injected trigger removed from the restored copy first: {bool(injected)}"
step("and forward again", forward)
finish()
