def cmp(a,b):
    strip=lambda rows,i: sorted([r[:i]+r[i+1:] for r in rows],key=repr)
    names=[r[1] for r in q("pragma table_info(transactions)")]; i=names.index("updated_at")
    expect(strip(a["transactions"],i)==strip(b["transactions"],i),"transactions differ")
    for t in ("transaction_tags","transaction_links","tags","accounts","budgets","statements","merchant_rules","custom_categories","category_meta"): 
        n=[r[1] for r in q(f"pragma table_info({t})")]
        A,B=(strip(a[t],n.index("updated_at")),strip(b[t],n.index("updated_at"))) if "updated_at" in n else (a[t],b[t]); expect(A==B,t+" differs")
